```ts
// HTML → plain text for news bodies (ARCHITECTURE §11.1.1 point 4). Pure functions; no DOM or Deno API.

/** The fixed list of named entities we decode (§11.1.1); any other named entity is kept as written. */
const NAMED_ENTITIES: Readonly<Record<string, string>> = {
  nbsp: '\u00a0',
  amp: '&',
  lt: '<',
  gt: '>',
  quot: '"',
  apos: "'",
  hellip: '\u2026',
  mdash: '\u2014',
  ndash: '\u2013',
  lsquo: '\u2018',
  rsquo: '\u2019',
  ldquo: '\u201c',
  rdquo: '\u201d',
  copy: '\u00a9',
  reg: '\u00ae',
  trade: '\u2122',
};
/** XML only predefines these five (plus numeric references). */
const XML_ENTITIES = new Set(['amp', 'lt', 'gt', 'quot', 'apos']);

const codePoint = (raw: string, cp: number): string =>
  Number.isInteger(cp) && cp > 0 && cp <= 0x10ffff && (cp < 0xd800 || cp > 0xdfff) ? String.fromCodePoint(cp) : raw;

function decode(s: string, allowed: (name: string) => boolean): string {
  return s.replace(/&(#[0-9]{1,7}|#[xX][0-9a-fA-F]{1,6}|[a-zA-Z][a-zA-Z0-9]{1,31});/g, (raw, body: string) => {
    if (body.startsWith('#x') || body.startsWith('#X')) return codePoint(raw, Number.parseInt(body.slice(2), 16));
    if (body.startsWith('#')) return codePoint(raw, Number.parseInt(body.slice(1), 10));
    const value = NAMED_ENTITIES[body];
    return value !== undefined && allowed(body) ? value : raw;
  });
}

/** Decodes XML's 5 predefined entities and numeric references (text outside CDATA). */
export const decodeXmlEntities = (s: string): string => decode(s, (name) => XML_ENTITIES.has(name));
/** Decodes numeric references and the fixed named list; unknown entities stay as written. */
export const decodeHtmlEntities = (s: string): string => decode(s, () => true);

const VOID = new Set(['area', 'base', 'br', 'col', 'embed', 'hr', 'img', 'input', 'link', 'meta', 'param', 'source', 'track', 'wbr']);
const SKIP = new Set(['script', 'style', 'noscript']);
/** Block-level tags that implicitly end an open <p>. */
const BLOCK = new Set(['p', 'div', 'ul', 'ol', 'li', 'table', 'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'blockquote', 'section', 'article', 'figure', 'header', 'footer', 'pre', 'form', 'aside', 'nav']);

const TOKEN = /<!--[\s\S]*?-->|<!\[CDATA\[[\s\S]*?\]\]>|<[!?][^>]*>|<(\/?)([a-zA-Z][a-zA-Z0-9:-]*)((?:[^>"']|"[^"]*"|'[^']*')*)>/g;

const hasClass = (attrs: string, cls: string): boolean => {
  const m = /(?:^|\s)class\s*=\s*(?:"([^"]*)"|'([^']*)')/i.exec(attrs);
  const value = m?.[1] ?? m?.[2] ?? '';
  return value.split(/\s+/).includes(cls);
};

/** Normalises one paragraph: decode entities, collapse whitespace (incl. NBSP), trim, NFC. */
const cleanParagraph = (raw: string): string => decodeHtmlEntities(raw).replace(/\s+/g, ' ').trim().normalize('NFC');

/**
 * Returns the text of every <p> (in document order). With `containerClass`, only <p> inside an element whose class list
 * contains that token count; nested containers are not counted twice. <script>/<style>/<noscript> content is dropped.
 */
export function htmlParagraphs(html: string, containerClass: string | null): string[] {
  const stack: { name: string; container: boolean }[] = [];
  const out: string[] = [];
  let containerDepth = 0;
  let para: string | null = null;
  let last = 0;
  const flush = (): void => {
    if (para !== null) {
      const text = cleanParagraph(para);
      if (text !== '') out.push(text);
      para = null;
    }
  };
  const inScope = (): boolean => containerClass === null || containerDepth > 0;
  const lower = html.toLowerCase();
  TOKEN.lastIndex = 0;
  for (let m = TOKEN.exec(html); m !== null; m = TOKEN.exec(html)) {
    if (para !== null) para += html.slice(last, m.index);
    last = TOKEN.lastIndex;
    const name = m[2]?.toLowerCase();
    if (name === undefined) continue; // comment, CDATA, doctype, processing instruction
    const closing = m[1] === '/';
    const attrs = m[3] ?? '';
    if (!closing && SKIP.has(name)) {
      const end = lower.indexOf(`</${name}`, TOKEN.lastIndex);
      const close = end < 0 ? -1 : html.indexOf('>', end);
      TOKEN.lastIndex = close < 0 ? html.length : close + 1;
      last = TOKEN.lastIndex;
      continue;
    }
    if (name === 'br') {
      if (para !== null) para += ' ';
    } else if (BLOCK.has(name)) {
      flush();
    }
    if (closing) {
      const at = stack.map((e) => e.name).lastIndexOf(name);
      if (at >= 0) {
        for (const e of stack.splice(at)) if (e.container) containerDepth -= 1;
        if (!inScope()) flush();
      }
      continue;
    }
    if (VOID.has(name) || attrs.trimEnd().endsWith('/')) continue;
    const container = containerClass !== null && hasClass(attrs, containerClass);
    if (container) containerDepth += 1;
    stack.push({ name, container });
    if (name === 'p' && inScope()) para = '';
  }
  if (para !== null) para += html.slice(last);
  flush();
  return out;
}

/** Body text = paragraphs joined by a blank line (§11.1.1). */
export const joinParagraphs = (paragraphs: readonly string[]): string => paragraphs.join('\n\n');

/** Word count = number of whitespace-separated tokens (threshold ≥ 250, §11.1.1). */
export const wordCount = (body: string): number => body.split(/\s+/).filter((w) => w !== '').length;
```

```ts
// Minimal RSS 2.0 reader (ARCHITECTURE §11.1.1 point 4). Pure: string in, Result out; no DOM or Deno API.
import { err, ok, type NewsCategory, type Result } from '@english-coach/contracts';
import { decodeXmlEntities } from './html.ts';
import { unavailable, type NewsUnavailable } from './types.ts';

export type RssItem = {
  readonly title: string;
  readonly link: string;
  readonly publishedAt: Date;
  readonly categories: readonly string[];
  /** Longest <content:encoded>, else <description>, else ''. Still HTML. */
  readonly bodyHtml: string;
};

export const MAX_ITEMS = 100;

const MONTHS = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];
const RFC822 =
  /^(?:[A-Za-z]{3},\s*)?(\d{1,2})\s+([A-Za-z]{3})\s+(\d{4})\s+(\d{2}):(\d{2})(?::(\d{2}))?\s+(?:([+-])(\d{2})(\d{2})|GMT|UTC|UT|Z)$/;

/** RFC 822 date as used by RSS 2.0 (numeric offset or GMT／UT／UTC／Z). Anything else → null (the item is skipped). */
export function parseRfc822(raw: string): Date | null {
  const m = RFC822.exec(raw.trim());
  if (m === null) return null;
  const [, d, mon, y, hh, mm, ss, sign, oh, om] = m;
  const month = MONTHS.indexOf((mon ?? '').toLowerCase());
  const day = Number(d);
  const hour = Number(hh);
  const minute = Number(mm);
  const second = Number(ss ?? '0');
  if (month < 0 || day < 1 || day > 31 || hour > 23 || minute > 59 || second > 60) return null;
  const offsetMin = sign === undefined ? 0 : (sign === '-' ? -1 : 1) * (Number(oh) * 60 + Number(om));
  const midnight = Date.UTC(Number(y), month, day);
  if (new Date(midnight).getUTCDate() !== day) return null; // e.g. 31 Feb
  return new Date(midnight + ((hour * 60 + minute) * 60 + second) * 1000 - offsetMin * 60_000);
}

const CDATA = /<!\[CDATA\[([\s\S]*?)\]\]>/g;
const MARK = '\u0000';

/** Text of an element: CDATA kept as is, everything else XML-entity-decoded. */
const textOf = (inner: string, cdata: readonly string[]): string =>
  inner
    .split(new RegExp(`${MARK}(\\d+)${MARK}`))
    .map((part, i) => (i % 2 === 1 ? (cdata[Number(part)] ?? '') : decodeXmlEntities(part)))
    .join('');

const elements = (xml: string, tag: string): string[] => {
  const re = new RegExp(`<${tag}(?:\\s[^>]*)?(?:/>|>([\\s\\S]*?)</${tag}\\s*>)`, 'g');
  return [...xml.matchAll(re)].map((m) => m[1] ?? '');
};

/** Parses an RSS 2.0 feed (UTF-8 only). DOCTYPE／ENTITY declarations are dropped and never expanded. */
export function parseRss(xml: string, category: NewsCategory): Result<RssItem[], NewsUnavailable> {
  const decl = /^\uFEFF?\s*<\?xml\b([^?]*)\?>/.exec(xml);
  const encoding = decl === null ? null : /\bencoding\s*=\s*["']([^"']+)["']/i.exec(decl[1] ?? '');
  if (encoding !== null && encoding[1]?.toLowerCase() !== 'utf-8') return err(unavailable(category, 'rss: not utf-8'));
  const cdata: string[] = [];
  const body = xml
    .replace(CDATA, (_m, inner: string) => `${MARK}${String(cdata.push(inner) - 1)}${MARK}`)
    .replace(/<!DOCTYPE(?:[^[>]|\[[\s\S]*?\])*>/gi, '')
    .replace(/<!--[\s\S]*?-->/g, '');
  if (!/<rss\b[^>]*\bversion\s*=\s*["']2\.0["']/.test(body)) return err(unavailable(category, 'rss: not rss 2.0'));
  const items: RssItem[] = [];
  for (const inner of elements(body, 'item').slice(0, MAX_ITEMS)) {
    const one = (tag: string): string => textOf(elements(inner, tag)[0] ?? '', cdata).trim();
    const publishedAt = parseRfc822(one('pubDate'));
    const title = one('title').replace(/\s+/g, ' ');
    const link = one('link');
    if (publishedAt === null || title === '' || link === '') continue;
    const encoded = elements(inner, 'content:encoded').map((e) => textOf(e, cdata));
    const longest = encoded.reduce((a, b) => (b.length > a.length ? b : a), '');
    items.push({
      title,
      link,
      publishedAt,
      categories: elements(inner, 'category').map((e) => textOf(e, cdata).trim()),
      bodyHtml: longest !== '' ? longest : textOf(elements(inner, 'description')[0] ?? '', cdata),
    });
  }
  return ok(items);
}
```

```ts
// robots.txt per RFC 9309 (ARCHITECTURE §11.1.1 point 4). Pure functions.

export const ROBOTS_PRODUCT_TOKEN = 'english-coach';

export type RobotsRules = { readonly allow: readonly string[]; readonly disallow: readonly string[] };
export const ALLOW_ALL: RobotsRules = { allow: [], disallow: [] };
export const DISALLOW_ALL: RobotsRules = { allow: [], disallow: ['/'] };

type Group = { agents: string[]; allow: string[]; disallow: string[] };

/** Rules of the groups whose user-agent equals our product token (case-insensitive), else of the `*` groups. */
export function parseRobots(txt: string, token: string = ROBOTS_PRODUCT_TOKEN): RobotsRules {
  const groups: Group[] = [];
  let current: Group | null = null;
  let inAgents = false;
  for (const rawLine of txt.split(/\r\n|\r|\n/)) {
    const line = (rawLine.split('#')[0] ?? '').trim();
    const colon = line.indexOf(':');
    if (colon < 0) continue;
    const key = line.slice(0, colon).trim().toLowerCase();
    const value = line.slice(colon + 1).trim();
    if (key === 'user-agent') {
      if (current === null || !inAgents) {
        current = { agents: [], allow: [], disallow: [] };
        groups.push(current);
      }
      current.agents.push(value.toLowerCase());
      inAgents = true;
    } else if (key === 'allow' || key === 'disallow') {
      inAgents = false;
      if (current === null || value === '') continue;
      (key === 'allow' ? current.allow : current.disallow).push(value);
    }
  }
  const own = groups.filter((g) => g.agents.includes(token.toLowerCase()));
  const chosen = own.length > 0 ? own : groups.filter((g) => g.agents.includes('*'));
  return { allow: chosen.flatMap((g) => g.allow), disallow: chosen.flatMap((g) => g.disallow) };
}

const matches = (pattern: string, path: string): boolean => {
  const anchored = pattern.endsWith('$');
  const body = anchored ? pattern.slice(0, -1) : pattern;
  const source = body
    .split('*')
    .map((s) => s.replace(/[.+?^${}()|[\]\\]/g, '\\$&'))
    .join('.*');
  return new RegExp(`^${source}${anchored ? '$' : ''}`).test(path);
};

/** Longest matching rule wins; on equal length Allow wins; no match → allowed. `pathAndQuery` starts with `/`. */
export function isAllowed(rules: RobotsRules, pathAndQuery: string): boolean {
  const longest = (patterns: readonly string[]): number =>
    patterns.filter((p) => matches(p, pathAndQuery)).reduce((n, p) => Math.max(n, p.length), -1);
  const allow = longest(rules.allow);
  const disallow = longest(rules.disallow);
  return disallow < 0 || allow >= disallow;
}
```

```ts
// globalThis.fetch). Guards from ARCHITECTURE §11.1.1 point 4: timeout, streamed size limit, content-type, manual redirects.
import { err, ok, type NewsCategory, type Result } from '@english-coach/contracts';
import { unavailable, type NewsUnavailable } from './types.ts';

export type FetchLike = (url: string, init: RequestInit) => Promise<Response>;
export type FetchKind = 'feed' | 'page' | 'robots';

export const USER_AGENT = 'english-coach personal reader';
export const TIMEOUT_MS = 10_000;
export const MAX_BYTES: Readonly<Record<FetchKind, number>> = { feed: 2 * 1024 * 1024, page: 1024 * 1024, robots: 256 * 1024 };
export const MAX_REDIRECTS = 3;

export type HttpText = { readonly status: number; readonly url: string; readonly text: string };

export type HttpOptions = {
  readonly fetch: FetchLike;
  readonly category: NewsCategory;
  readonly timeoutMs: number;
  /** Checked for the first URL and again for every redirect target (§7: only news_source URLs). */
  readonly allowUrl: (url: string) => boolean;
};

const contentTypeOk = (kind: FetchKind, type: string | null): boolean => {
  if (kind === 'robots') return true;
  const t = (type ?? '').toLowerCase();
  return kind === 'feed' ? t.includes('xml') || t.includes('rss') : t.includes('html');
};

async function readLimited(res: Response, limit: number): Promise<Uint8Array | null> {
  if (res.body === null) return new Uint8Array(0);
  const reader = res.body.getReader();
  const chunks: Uint8Array[] = [];
  let size = 0;
  for (let r = await reader.read(); !r.done; r = await reader.read()) {
    size += r.value.byteLength;
    if (size > limit) {
      await reader.cancel();
      return null;
    }
    chunks.push(r.value);
  }
  const out = new Uint8Array(size);
  let at = 0;
  for (const c of chunks) {
    out.set(c, at);
    at += c.byteLength;
  }
  return out;
}

const discard = async (res: Response): Promise<void> => {
  await res.body?.cancel().catch(() => undefined);
};

/**
 * GET with our User-Agent. 2xx → ok({ status, url, text }); other final statuses → ok with text ''.
 * Network error, timeout, body over the limit, non-UTF-8 body, wrong content-type, or a redirect that leaves the allowed
 * URLs (or more than 3 redirects) → err(news_unavailable) with a short reason code.
 */
export async function getText(url: string, kind: FetchKind, o: HttpOptions): Promise<Result<HttpText, NewsUnavailable>> {
  const signal = AbortSignal.timeout(o.timeoutMs);
  const fail = (reason: string): Result<HttpText, NewsUnavailable> =>
    err(unavailable(o.category, `${kind}: ${signal.aborted ? 'timeout' : reason}`));
  let current = url;
  for (let hop = 0; hop <= MAX_REDIRECTS; hop += 1) {
    if (!o.allowUrl(current)) return fail('url not allowed');
    let res: Response;
    try {
      res = await o.fetch(current, {
        method: 'GET',
        redirect: 'manual',
        signal,
        headers: { 'User-Agent': USER_AGENT, Accept: kind === 'feed' ? 'application/rss+xml, application/xml' : '*/*' },
      });
    } catch {
      return fail('network');
    }
    if (res.status >= 300 && res.status < 400) {
      const location = res.headers.get('location');
      await discard(res);
      if (location === null) return fail('redirect without location');
      try {
        current = new URL(location, current).toString();
      } catch {
        return fail('bad redirect');
      }
      continue;
    }
    if (res.type === 'opaqueredirect' || res.status === 0) return fail('opaque redirect');
    if (res.status < 200 || res.status >= 300) {
      await discard(res);
      return ok({ status: res.status, url: current, text: '' });
    }
    if (!contentTypeOk(kind, res.headers.get('content-type'))) {
      await discard(res);
      return fail('content-type');
    }
    let bytes: Uint8Array | null;
    try {
      bytes = await readLimited(res, MAX_BYTES[kind]);
    } catch {
      return fail('network');
    }
    if (bytes === null) return fail('too large');
    try {
      return ok({ status: res.status, url: current, text: new TextDecoder('utf-8', { fatal: true }).decode(bytes) });
    } catch {
      return fail('not utf-8');
    }
  }
  return fail('too many redirects');
}
```

```ts
  const host = page.ok && page.value.status < 300 ? parseUrl(page.value.url)?.hostname : undefined;
  const body = page.ok && host !== undefined ? extractPageBody(page.value.text, host) : null;
  if (body === null) ctx.deps.log({ type: 'extract_failed', sourceId: ctx.source.id });
  return body;
}

type SourceResult = Result<Candidate | null, AppError>;

async function fromSource(ctx: Ctx): Promise<SourceResult> {
  const { source, deps } = ctx;
  const failed = (reason: string): SourceResult => {
    deps.log({ type: 'source_failed', sourceId: source.id, reason });
    return ok(null);
  };
  if (!(await robotsAllow(ctx, source.feedUrl))) return failed('robots');
  const feed = await get(ctx, source.feedUrl, 'feed');
  if (!feed.ok) return failed(feed.error.message);
  if (feed.value.status >= 300) return failed(`feed: http ${String(feed.value.status)}`);
  const parsed = parseRss(feed.value.text, source.category);
  if (!parsed.ok) return failed(parsed.error.message);
  const cutoff = deps.now.getTime() - deps.lookbackHours * 3_600_000;
  const items = parsed.value
    .filter((i) => i.publishedAt.getTime() >= cutoff && !isSponsored(i.categories))
    .sort((a, b) => b.publishedAt.getTime() - a.publishedAt.getTime());
  let judged = 0;
  for (const item of items) {
    const url = normalizeArticleUrl(item.link);
    if (url === null) continue;
    const patternOk = urlPatternOk(source.includeUrlPattern, url);
    if (patternOk === null) return failed('bad include_url_pattern');
    if (!patternOk) continue;
    const body = await bodyOf(ctx, item, url);
    if (body === null) continue;
    const words = wordCount(body);
    if (words < MIN_WORDS) continue;
    const sha256 = await sha256Hex(body);
    const used = await deps.isUsed({ url, sha256 });
    if (!used.ok) return used;
    if (used.value) continue;
    const candidate: Candidate = {
      sourceId: source.id,
      category: source.category,
      url,
      title: item.title,
      publishedAt: item.publishedAt,
      body,
      sha256,
      wordCount: words,
    };
    const match = await deps.judge(candidate);
    if (!match.ok) return match;
    if (match.value) return ok(candidate);
    judged += 1;
    if (judged >= MAX_JUDGED_PER_SOURCE) return ok(null);
  }
  return ok(null);
}

/**
 * Tries the enabled sources of `category` by priority 1 → 2 → 3 and returns the first accepted article.
 * A failing source (HTTP error, timeout, robots, bad feed) counts as "no news" and the next source is tried.
 * No source has one → err(news_unavailable) so daily-prep moves on to the next category (§11.2 step 2).
 * Errors from isUsed／judge (DB, provider) are returned as they are.
 */
export async function findArticle(
  category: NewsCategory,
  sources: readonly NewsSource[],
  deps: IngestDeps,
): Promise<Result<Candidate, AppError>> {
  const ordered = sources.filter((s) => s.enabled && s.category === category).sort((a, b) => a.priority - b.priority);
  for (const source of ordered) {
    const r = await fromSource({ source, deps });
    if (!r.ok) return r;
    if (r.value !== null) return ok(r.value);
  }
  return err(unavailable(category, 'no source had a usable article'));
}
```

```ts
/** Injected fetch over a fixed route table; unknown URLs get 404. Records every call. */
export function fakeFetch(routes: Readonly<Record<string, Route>>): {
  fetch: (url: string, init: RequestInit) => Promise<Response>;
  calls: { url: string; init: RequestInit }[];
} {
  const calls: { url: string; init: RequestInit }[] = [];
  const fetch = (url: string, init: RequestInit): Promise<Response> => {
    calls.push({ url, init });
    const r = routes[url];
    const headers = new Headers();
    if (r?.type !== undefined) headers.set('content-type', r.type);
    if (r?.location !== undefined) headers.set('location', r.location);
    return Promise.resolve(new Response(r?.body ?? '', { status: r?.status ?? 404, headers }));
  };
  return { fetch, calls };
}
```
