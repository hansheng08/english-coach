`packages/domain/src/plan.ts`（輸入型別在本檔定義，domain 內部用、不跨網路）：
```ts
export type CardCandidate = { readonly cardId: CardId; readonly schedule: CardSchedule; readonly priorityNew: boolean; readonly createdAtMs: number };
export type SeedCandidate = { readonly seedChunkId: string; readonly batch: number };
export type DailyCardSelection = { readonly oldCardIds: readonly CardId[]; readonly newCardIds: readonly CardId[]; readonly seedChunkIds: readonly string[]; readonly extraOldCardIds: readonly CardId[]; readonly total: number };
export function selectDailyCards(input: { readonly cards: readonly CardCandidate[]; readonly seeds: readonly SeedCandidate[]; readonly usedSeedChunkIds: ReadonlySet<string> }, today: PlanDate): DailyCardSelection;
export type QuestionCandidate = { readonly questionId: QuestionId; readonly source: 'resume' | 'bank'; readonly category: 'intro' | 'behavioral' | 'technical'; readonly retired: boolean; readonly timesAsked: number; readonly lastAskedDate: PlanDate | null };

export type ReviewedTodayCard = { readonly cardId: CardId; readonly chunk: string; readonly state: CardState; readonly intervalStep: IntervalStep; readonly reviewedAtMs: number };
export type ReviewingFallbackCard = { readonly cardId: CardId; readonly chunk: string; readonly intervalStep: IntervalStep; readonly lastReviewedDate: PlanDate };

export type ShadowSentence = { readonly source: 'news' | 'rewrite'; readonly text: string };
export function selectShadowSentences(newsSentences: readonly ShadowSentenceCandidate[], rewriteSentences: readonly ShadowSentenceCandidate[] | null): readonly ShadowSentence[];
```
行為照 §4.4～§4.6；以下是 v1.2 已寫明、本版沿用的細節與決定：
- `DailyCardSelection`：`oldCardIds`＝到期舊卡前 15 張；`newCardIds`＝新卡；`seedChunkIds`＝種子補位；`extraOldCardIds`＝新卡＋種子仍不足時，從到期舊卡第 16 張起補；`total`＝四者合計（≤ 25）。各陣列依排序。
- 排序的最後一鍵都是 id，字串比較用 `<`／`>`（不用 `localeCompare`）。`createdAt` 以 `createdAtMs` 表示，`cardId` 依字串比較。
- **跟讀句字數**：`newsWords(text).length` 在 6～25（含）才選（`newsWords` 從 `@english-coach/contracts` import；`/` 不算字，單獨的標點算 1 字）。**§4.6 的字數算法以 ADR-013 為準**（幕僚長 2026-11-25 21:45 裁定）。`normalizeWords` 不用於這個篩選。
- `containsTargetChunk` 由呼叫端依 `news_article.chunks` 的 `sentence_index` 或升級語塊標記算好傳入。
- **TechLead 決定**（v1.2 沿用；幕僚長 2026-11-23 裁定實作層級的選擇由 TechLead 決定）：
  1. `selectTonightQuestion` 不加 §4.5 簽章裡的 `today`（排序用不到，未使用的參數會被 lint 擋），報告的「設計決定」寫明。
  2. 晚上提示 chunk 的 fallback 排序（§4.5 沒寫死）：`lastReviewedDate` 晚 → `intervalStep` 大 → `cardId`，並排除已選的 `cardId`。
  3. `QuestionCandidate` 的 `source`、`category` 用本檔的字串 union，不新增 contracts schema。

## 驗收標準
- **AC-T10-1**（§4.6）`normalizeWords`：以空白切字、轉小寫、去掉頭尾不是 `a-z`／`0-9`／`'` 的字元，去掉後是空字串的丟掉（單獨的 `/`、`—`）；字中間的撇號保留。
- **AC-T10-2**（AC-07、§4.4）到期舊卡的篩選與排序：只取 `state` 為 collected／learning／reviewing、`dueDate` 不是 `null` 且 ≤ `today` 的卡；mastered 與未到期的不選；排序 `dueDate` 早 → `intervalStep` 小 → `createdAtMs` 小 → `cardId`。
- **AC-T10-3**（AC-07、Q-12、§4.4）舊卡上限：`oldCardIds` 最多 15 張；到期舊卡 0／10／15／30 張且新卡充足時，舊＋新依序為 0＋25、10＋15、15＋10、15＋10（30 張時不用第 16 張）。
- **AC-T10-4**（AC-07、AC-16、§4.4）新卡：只取 `state = collected` 且 `dueDate = null`；`priorityNew = true` 的排在前面，其次 `createdAtMs` 小、`cardId`；新卡＋種子合計最多 `25 − 舊卡數`，先新卡。
- **AC-T10-5**（AC-07、Q-22、§4.4）種子：`usedSeedChunkIds` 裡的種子不選；排序 `batch` 小 → `seedChunkId`；只在新卡不足時補。
- **AC-T10-6**（Q-22、§4.4）不足時補位：新卡＋種子仍不足 `25 − 舊卡數` 時，從到期舊卡第 16 張起依排序補進 `extraOldCardIds`；全部用完時 `total < 25`；`total` 等於四個陣列長度合計。
- **AC-T10-7**（ADR-011 決定 4）可重現：輸入陣列打亂順序，`selectDailyCards`、`selectTonightQuestion`、`selectEveningHintChunks` 的結果都不變。
- **AC-T10-8**（AC-13、Q-23、§4.5）今晚題目：`retired` 的不選；有任何 `source = 'resume'` 時只選履歷題，否則選題庫題；排序 `timesAsked` 少 → `lastAskedDate` 早（`null` 最前）→ `category` 依 `intro`、`behavioral`、`technical` → `questionId`；沒有候選回 `null`。
- **AC-T10-9**（AC-13、§4.5）晚上提示 chunk：今天評過的卡依 reviewing 優先 → `intervalStep` 大 → `reviewedAtMs` 小 → `cardId` 取前 3；不足時依 TechLead 決定 2 從 fallback 補、不重複；仍不足回實際數量。
- **AC-T10-10**（AC-05、§4.6）跟讀比例：`rewriteSentences = null` → 新聞 5 句；否則新聞 3＋改寫 2；一方不足由另一方補；合計不足回實際句數；輸出先新聞、後改寫，`source` 正確。
- **AC-T10-11**（AC-05、§4.6、ADR-013）跟讀篩選與排序：只選 `newsWords` 字數 6～25 的句子；各方內部 `containsTargetChunk = true` 優先，其餘照原陣列順序。
- **AC-T10-12**（CLAUDE.md §4）匯出：`@english-coach/domain` 匯出 `normalizeWords` 與介面列出的 4 個函式和 8 個型別。
