```sql
-- T-005 migration 2: news_source, news_article, resume, question, daily_plan + RLS.
-- Source: docs/ARCHITECTURE.md §1, §4 (tables), §4.5 (tonight question, bank questions), §5.1 (/plan/today, /resume), §7, §7.1, §11.2; ADR-006.
-- The original article text is never stored, only its SHA-256 (CLAUDE.md §5 #7). News rotation defaults live only in packages/contracts (CLAUDE.md §5 #11).
-- news_source is shared reference data (seeded by T-022) and bank questions (source = 'bank', imported like seed_chunk) are shared rows, so neither has an owner.

create table public.news_source (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  feed_url text not null constraint news_source_feed_url_key unique,
  category text not null constraint news_source_category_check check (category in ('ai', 'software', 'semiconductor')),
  priority smallint not null constraint news_source_priority_check check (priority between 1 and 3),
  fetch_mode text not null constraint news_source_fetch_mode_check check (fetch_mode in ('rss_full', 'rss_then_page')),
  include_url_pattern text,
  enabled boolean not null default true,
  terms_checked_at date,
  created_at timestamptz not null default now(),
  constraint news_source_category_priority_key unique (category, priority)
);

create table public.news_article (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users (id) on delete cascade,
  source_id uuid not null references public.news_source (id) on delete restrict,
  category text not null constraint news_article_category_check check (category in ('ai', 'software', 'semiconductor')),
  original_url text not null constraint news_article_original_url_key unique,
  original_sha256 text not null constraint news_article_original_sha256_key unique constraint news_article_original_sha256_check check (original_sha256 ~ '^[0-9a-f]{64}$'),
  title text not null,
  published_at timestamptz not null,
  rewritten_text text not null,
  chunks jsonb not null constraint news_article_chunks_check check (case when jsonb_typeof(chunks) = 'array' then jsonb_array_length(chunks) = 5 else false end),
  sentences jsonb not null constraint news_article_sentences_check check (jsonb_typeof(sentences) = 'array'),
  read_at timestamptz,
  created_at timestamptz not null default now(),
  constraint news_article_id_user_id_key unique (id, user_id)
);

create table public.resume (
  user_id uuid primary key references auth.users (id) on delete cascade,
  storage_path text not null constraint resume_storage_path_check check (starts_with(storage_path, user_id::text || '/')),
  redacted_text text not null,
  uploaded_at timestamptz not null default now()
);

create table public.question (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users (id) on delete cascade,
  resume_id uuid references public.resume (user_id) on delete set null,
  source text not null constraint question_source_check check (source in ('resume', 'bank')),
  category text not null constraint question_category_check check (category in ('intro', 'behavioral', 'technical')),
  text text not null,
  cited_experience text,
  retired boolean not null default false,
  created_at timestamptz not null default now(),
  constraint question_source_owner_check check (
    (source = 'bank' and user_id is null and resume_id is null)
    or (source = 'resume' and user_id is not null and cited_experience is not null)
  ),
  constraint question_resume_owner_check check (resume_id is null or resume_id = user_id),
  constraint question_retired_check check (source = 'bank' or resume_id is not null or retired)
);

create table public.daily_plan (
  user_id uuid not null references auth.users (id) on delete cascade,
  plan_date date not null,
  planned_category text not null constraint daily_plan_planned_category_check check (planned_category in ('ai', 'software', 'semiconductor')),
  actual_category text constraint daily_plan_actual_category_check check (actual_category in ('ai', 'software', 'semiconductor')),
  category_skipped text[] not null default '{}' constraint daily_plan_category_skipped_check check (category_skipped <@ array['ai', 'software', 'semiconductor']::text[]),
  article_id uuid,
  is_fallback_article boolean not null default false,
  shadow_items jsonb not null default '[]'::jsonb constraint daily_plan_shadow_items_check check (case when jsonb_typeof(shadow_items) = 'array' then jsonb_array_length(shadow_items) <= 5 else false end),
```

## TechLead 決定（ARCHITECTURE／ADR 沒有寫死的實作細節；照做，不需要再問）
1. **哪些是使用者資料、哪些是共用列**：§4「所有表都有 `user_id`」照 T-004 決定 1 解讀為「使用者資料表」。`news_article`、`resume`、`daily_plan` 與履歷題（`question.source = 'resume'`）都是執行期替使用者產生的資料，一律 `user_id uuid not null references auth.users (id) on delete cascade`（`resume.user_id` 就是 PK，§4「一次一份」）。`news_source`（T-022 以 migration 匯入 §11.1 的 9 個來源）與題庫題（§4.5「比照種子庫…以 migration 匯入」、Q-23）在 migration 執行時還沒有任何使用者，所以和 T-004 的 `seed_chunk` 一樣是共用列：`news_source` 沒有 `user_id`；`question.user_id` 可為 null，`question_source_owner_check` 規定 `bank` 題 `user_id`、`resume_id` 都是 null，`resume` 題一定有 `user_id` 與 `cited_experience`（AC-11「每題標示引用的履歷經歷」）。題庫和履歷題放同一張表，因為 §4 `question.source` 就是 `resume`／`bank`、§4.5 從同一批候選選題。
2. **權限矩陣**（RLS 一律 `user_id = (select auth.uid())`，只開給 `authenticated`；`anon` 對 5 張表全部 `revoke all`；`service_role` 全部權限、繞過 RLS，§7.1 只給 `daily-prep`／`purge-audio`）。§1 寫明排程失敗時 App 會呼叫同一支冪等的 `POST /plan/today` 補做，而 §4、§7.1 規定 `api` 一律以使用者 JWT 查表，所以補做時要寫入的表必須對 `authenticated` 開放（只限自己的列）：
   - `news_source`：SELECT（補做時讀白名單；`enabled` 由 Hansheng 在 Supabase 後台或之後的 migration 改，ADR-006；本單不開寫入）。
   - `news_article`：SELECT、INSERT，**UPDATE 只開 `read_at` 一欄**（§11.2 第 3 步「最近一篇未讀」；文章內容寫入後不可改）；沒有 DELETE。
   - `resume`：SELECT、INSERT、UPDATE、DELETE（`POST /resume` 新的取代舊的、`DELETE /resume`、AC-12、Q-09）。
   - `question`：SELECT（自己的＋題庫題）、INSERT（只能新增自己的履歷題）、**UPDATE 只開 `retired` 一欄**（取代或刪除履歷時標 retired）；沒有 DELETE（題目保留，之後的 `evening_answer` 會引用，T-006）；題庫題對使用者唯讀。
   - `daily_plan`：SELECT、INSERT、UPDATE（補做、`status` 更新、冪等重做）；沒有 DELETE。
3. **跨使用者參照**：`daily_plan (article_id, user_id) → news_article (id, user_id)` 複合外鍵（`news_article` 加 `unique (id, user_id)`），計畫只能用自己的文章（secret 金鑰寫入時也成立）。`tonight_question_id` 可能指向沒有 `user_id` 的題庫題，不能用複合外鍵，改為一般外鍵＋trigger `daily_plan_question_guard`（`set search_path = ''`、不用 `security definer`）：題目必須是題庫題或計畫擁有者自己的，否則 `23503`。`question.resume_id` 參照 `resume (user_id)`（`resume` 的 PK）＋`question_resume_owner_check (resume_id = user_id)`。`card_ids uuid[]` 是陣列、不能加外鍵，元素屬於擁有者由之後的 `daily-prep`／`POST /plan/today`（T-025）保證，本單只限制 ≤ 25 個。
4. **刪履歷**（§4「刪履歷時設 null 並標 `retired=true`」、§5.1 `DELETE /resume`、AC-12）：`question.resume_id … on delete set null`＋`question_retired_check`（履歷題沒有 `resume_id` 時一定要 `retired`），所以 `api` 必須先把該履歷的題目標 `retired`，再刪 `resume` 列；順序反了資料庫回 `23514`，題目不會被刪掉。`resume.storage_path` 必須在 `<user_id>/` 下（`resume_storage_path_check`，對應 `StoragePathSchema`）；Storage 原檔的刪除由 T-036 處理。
5. **`news_article` 不存原文、以 URL＋SHA-256 去重**（§4、ADR-006 第 3、4 點、CLAUDE.md §5 第 7 點）：沒有任何原文欄位；`original_url` 唯一；原文雜湊欄位名稱為 `original_sha256`（`text`，64 個小寫十六進位字元，唯一）。`chunks`（§4 寫 `jsonb[5]`）存成**一個 JSON 陣列**、剛好 5 個元素（AC-02「已標出 5 個語塊」）；`sentences` 是 JSON 陣列；元素格式由之後的 contracts 單用 zod 定義。`published_at` 必填（`lookback_hours` 篩選要用）；`rewritten_text` 的 250～300 字由應用層檢查，SQL 不寫字數。`original_url`／`original_sha256` 依單使用者規格（Q-01）採全表唯一。
6. **`daily_plan`**：主鍵 `(user_id, plan_date)`（§4「`user_id`＋`plan_date` 唯一」；contracts 沒有 `PlanId`，不另加 `id`）；`plan_date date`（台北日期，§4）；`planned_category` 必填、`actual_category` 可為 null（pending／failed）；`category_skipped text[]` 只能是 3 個類別；`shadow_items`（§4 寫 `jsonb[5]`）存成 JSON 陣列、最多 5 個（§4.6 不足時回傳實際句數）；`card_ids uuid[]` 最多 25 個（Q-22 可少於 25）；`is_fallback_article` 預設 false；`status` 預設 `pending`；`daily_plan_status_consistency_check`：`pending` 沒有 `error`；`ready` 一定有 `article_id`、`actual_category` 且沒有 `error`；`failed` 一定有 `error`（§11.2 第 4 步）。`tonight_question_id` 可為 null（沒有任何題目時 T-010 `selectTonightQuestion` 回 `null`）。
7. **`news_source`**：`priority` 1～3、`(category, priority)` 唯一（每類 1 主 2 備，§11.1）、`feed_url` 唯一、`fetch_mode` 只有 `rss_full`／`rss_then_page`、`include_url_pattern` 可為 null、`enabled` 預設 true（ADR-006「備用來源預設啟用」）、`terms_checked_at date` 可為 null（記錄最後檢查日）。類別字面值和 contracts 的 `NEWS_CATEGORIES`（`ai`、`software`、`semiconductor`）相同；SQL 不寫 `rotation_order`、`anchor_date`、`lookback_hours` 等輪替預設（CLAUDE.md §5 第 11 點）。
