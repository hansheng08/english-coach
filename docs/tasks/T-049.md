### 1. `packages/contracts/src/settings.ts`（修改，整檔 Write）
整檔逐字寫出：`SrsSettingsSchema` 與 `main` 完全相同；只改第 1 行註解、加 1 行 `import`，檔尾加 `NewsSettingsSchema`（TechLead 決定 2、3）。
```ts
// Schemas for the `settings.*` JSON columns: SrsSettingsSchema (T-008) and NewsSettingsSchema (T-049).
import { z } from 'zod';
import { NEWS_CATEGORIES, NewsCategorySchema, PlanDateSchema, type NewsCategory } from './ids.ts';

const DaysSchema = z.number().int().min(1).max(365);

/** 間隔複習設定（ARCHITECTURE §4.2）；預設值只寫在這裡，其他程式一律從設定讀。 */
export const SrsSettingsSchema = z
  .object({
    step_days: z.tuple([DaysSchema, DaysSchema, DaysSchema]).default([3, 7, 14]),
    after_last_days: DaysSchema.default(14),
    forgot_days: DaysSchema.default(1),
  })
  .superRefine((s, ctx) => {
    const [first, second, third] = s.step_days;
    if (second <= first) {
      ctx.addIssue({ code: 'custom', path: ['step_days', 1], message: 'step_days must be strictly increasing' });
    }
    if (third <= second) {
      ctx.addIssue({ code: 'custom', path: ['step_days', 2], message: 'step_days must be strictly increasing' });
    }
    if (s.after_last_days < third) {
      ctx.addIssue({
        code: 'custom',
        path: ['after_last_days'],
        message: 'after_last_days must be >= step_days[2]',
      });
    }
    if (s.forgot_days > first) {
      ctx.addIssue({ code: 'custom', path: ['forgot_days'], message: 'forgot_days must be <= step_days[0]' });
    }
  });

export type SrsSettings = z.infer<typeof SrsSettingsSchema>;

/** 每次 parse 都回新的陣列（預設順序＝NEWS_CATEGORIES，不另寫字串）。 */
const defaultRotationOrder = (): [NewsCategory, NewsCategory, NewsCategory] => [
  NEWS_CATEGORIES[0],
  NEWS_CATEGORIES[1],
  NEWS_CATEGORIES[2],
];

/** 每日新聞類別輪替設定（ARCHITECTURE §11.2、ADR-011 決定 6）；預設值只寫在這裡，其他程式一律從設定讀。 */
export const NewsSettingsSchema = z
  .object({
    rotation_order: z
      .tuple([NewsCategorySchema, NewsCategorySchema, NewsCategorySchema])
      .default(defaultRotationOrder),
    anchor_date: PlanDateSchema.prefault('2026-11-22'),
    lookback_hours: z.number().int().min(1).max(168).default(48),
  })
  .superRefine((s, ctx) => {
    if (new Set(s.rotation_order).size !== s.rotation_order.length) {
      ctx.addIssue({ code: 'custom', path: ['rotation_order'], message: 'rotation_order must list each category once' });
    }
  });

export type NewsSettings = z.infer<typeof NewsSettingsSchema>;
```

### 2. `packages/contracts/src/news-settings.test.ts`（新增）
`NewsSettingsSchema` 的 3 個測試（TechLead 決定 1、6）。
```ts
import { describe, expect, it } from 'vitest';
import { NewsSettingsSchema } from './settings.ts';

const issuePaths = (input: unknown): string[] => {
  const r = NewsSettingsSchema.safeParse(input);
  return r.success ? [] : r.error.issues.map((i) => i.path.map(String).join('.'));
};

describe('NewsSettingsSchema (ARCHITECTURE §11.2, ADR-011 decision 6)', () => {
  it('(AC-T49-1) parse({}) gives the documented defaults', () => {
    expect(NewsSettingsSchema.parse({})).toEqual({
      rotation_order: ['ai', 'software', 'semiconductor'],
      anchor_date: '2026-11-22',
      lookback_hours: 48,
    });
  });

  it('(AC-T49-1) accepts lookback_hours 1 and 168 and another rotation order', () => {
    expect(NewsSettingsSchema.safeParse({ lookback_hours: 1 }).success).toBe(true);
    expect(NewsSettingsSchema.safeParse({ lookback_hours: 168 }).success).toBe(true);
    expect(NewsSettingsSchema.parse({ rotation_order: ['semiconductor', 'ai', 'software'] }).rotation_order).toEqual([
      'semiconductor',
      'ai',
      'software',
    ]);
  });

  it('(AC-T49-1) rejects a repeated category (path rotation_order) and out-of-range values', () => {
    expect(issuePaths({ rotation_order: ['ai', 'ai', 'software'] })).toEqual(['rotation_order']);
    const rejects: unknown[] = [
      { rotation_order: ['ai', 'software'] },
      { rotation_order: ['ai', 'software', 'finance'] },
      { lookback_hours: 0 },
      { lookback_hours: 169 },
      { lookback_hours: 1.5 },
      { anchor_date: '2026-02-30' },
    ];
    expect(rejects.map((input) => NewsSettingsSchema.safeParse(input).success)).toEqual(rejects.map(() => false));
  });
});
```

```ts
import type { NewsCategory, NewsSettings, PlanDate } from '@english-coach/contracts';
import { daysBetween } from './taipei-date.ts';

export type RotationOrder = NewsSettings['rotation_order'];

/** rotation_order 往後數 offset 格（可為負數），三格一圈。 */
function stepFrom(order: RotationOrder, offset: number): NewsCategory {
  const i = ((offset % 3) + 3) % 3;
  if (i === 0) return order[0];
  if (i === 1) return order[1];
  return order[2];
}

/** 今天輪到的類別：rotation_order[(planDate − anchor_date 的天數) mod 3]；只看日期，錨點日之前也對（ARCHITECTURE §11.2）。 */
export function plannedCategory(
  planDate: PlanDate,
  settings: Pick<NewsSettings, 'rotation_order' | 'anchor_date'>,
): NewsCategory {
  return stepFrom(settings.rotation_order, daysBetween(settings.anchor_date, planDate));
}

/** 從 planned 開始依 rotation_order 往後繞一圈的 3 個類別（§11.2 備援第 2 步）。rotationOrder 必須已通過 NewsSettingsSchema。 */
export function categoryTryOrder(planned: NewsCategory, rotationOrder: RotationOrder): readonly NewsCategory[] {
  const start = rotationOrder.indexOf(planned);
  return [stepFrom(rotationOrder, start), stepFrom(rotationOrder, start + 1), stepFrom(rotationOrder, start + 2)];
}

export type NewsCategoryDecision = {
  readonly planned: NewsCategory;
  readonly actual: NewsCategory | null;
  readonly skipped: readonly NewsCategory[];
  readonly useFallbackArticle: boolean;
};

/** 依 categoryTryOrder 找第一個有新聞的類別；前面沒有的依序記入 skipped；三類都沒有 → 改用舊文章（§11.2 備援第 3 步）。 */
export function decideNewsCategory(
  planned: NewsCategory,
  rotationOrder: RotationOrder,
  hasNews: ReadonlySet<NewsCategory>,
): NewsCategoryDecision {
  const skipped: NewsCategory[] = [];
  for (const c of categoryTryOrder(planned, rotationOrder)) {
    if (hasNews.has(c)) return { planned, actual: c, skipped, useFallbackArticle: false };
    skipped.push(c);
  }
  return { planned, actual: null, skipped, useFallbackArticle: true };
}
```
